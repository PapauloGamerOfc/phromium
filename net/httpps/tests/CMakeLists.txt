cmake_minimum_required(VERSION 3.22)
project(phromium_httpps_native_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(CHROMIUM_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../apply_boringssl_patch.py"
    COMMAND_ERROR_IS_FATAL ANY)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPENSSL_NO_ASM ON CACHE BOOL "" FORCE)
add_subdirectory("${CHROMIUM_ROOT}/third_party/boringssl/src" boringssl EXCLUDE_FROM_ALL)
add_executable(httpps_native_test
    native_test.cc ../draft01.cc ../lab_pki.cc)
target_compile_definitions(httpps_native_test PRIVATE PHROMIUM_HTTPPS_STANDALONE)
target_include_directories(httpps_native_test PRIVATE "${CHROMIUM_ROOT}")
target_link_libraries(httpps_native_test PRIVATE crypto)
add_executable(httpps_pki_test pki_test.cc)
target_include_directories(httpps_pki_test PRIVATE "${CHROMIUM_ROOT}")
target_link_libraries(httpps_pki_test PRIVATE pki)
add_executable(httpps_pki_chain_test pki_chain_test.cc)
target_include_directories(httpps_pki_chain_test PRIVATE "${CHROMIUM_ROOT}")
target_link_libraries(httpps_pki_chain_test PRIVATE pki)
if(WIN32)
  target_link_libraries(httpps_native_test PRIVATE ws2_32)
endif()
enable_testing()
add_test(NAME tls20_primitives COMMAND httpps_native_test)
add_test(NAME httpps_spki_and_https_policy COMMAND httpps_pki_test)
add_test(NAME httpps_pki_path_builder COMMAND httpps_pki_chain_test)
