cmake_minimum_required(VERSION 3.22)
project(phromium_httpps_native_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Optional helper is built from the pinned reference-server checkout, never a
# downloaded binary or a second fork of the password crypto implementation.
set(HTTPPS_PAKE_SOURCE "" CACHE FILEPATH "Explicit tls20-server/native/pake.c")
if(HTTPPS_PAKE_SOURCE)
  file(SHA256 "${HTTPPS_PAKE_SOURCE}" HTTPPS_PAKE_ACTUAL_SHA256)
  if(NOT HTTPPS_PAKE_ACTUAL_SHA256 STREQUAL "a4d0a29de7cd7a7c6e816ca94e6d43947548d57a26ee82d56a88a8861fa60ebb")
    message(FATAL_ERROR "PAKE source does not match the reviewed pinned helper")
  endif()
  add_library(httpps_password_helper SHARED "${HTTPPS_PAKE_SOURCE}")
  set_target_properties(httpps_password_helper PROPERTIES
      OUTPUT_NAME tls20-pake C_STANDARD 11 C_STANDARD_REQUIRED ON
      C_EXTENSIONS OFF C_VISIBILITY_PRESET hidden)
  if(MSVC)
    target_compile_options(httpps_password_helper PRIVATE /W4 /WX /O2)
  else()
    target_compile_options(httpps_password_helper PRIVATE -O2 -Wall -Wextra -Werror)
  endif()
endif()
get_filename_component(CHROMIUM_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../apply_boringssl_patch.py"
    COMMAND_ERROR_IS_FATAL ANY)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPENSSL_NO_ASM ON CACHE BOOL "" FORCE)
add_subdirectory("${CHROMIUM_ROOT}/third_party/boringssl/src" boringssl EXCLUDE_FROM_ALL)
set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${CHROMIUM_ROOT}/third_party/zlib" zlib EXCLUDE_FROM_ALL)
file(GLOB BROTLI_SOURCES CONFIGURE_DEPENDS
    "${CHROMIUM_ROOT}/third_party/brotli/common/*.c"
    "${CHROMIUM_ROOT}/third_party/brotli/dec/*.c"
    "${CHROMIUM_ROOT}/third_party/brotli/enc/*.c")
add_library(httpps_brotli STATIC ${BROTLI_SOURCES})
target_include_directories(httpps_brotli PUBLIC "${CHROMIUM_ROOT}/third_party/brotli/include")
set(ZSTD_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZSTD_MULTITHREAD_SUPPORT OFF CACHE BOOL "" FORCE)
add_subdirectory("${CHROMIUM_ROOT}/third_party/zstd/src/build/cmake" zstd EXCLUDE_FROM_ALL)
add_executable(httpps_native_test
    tls_features_test.cc
    native_test.cc pinning_test.cc control_profiles_test.cc integrity_only_test.cc native_provider_test.cc ../draft01.cc ../lab_pki.cc ../application_auth.cc ../pinning.cc ../bootstrap.cc ../control_profiles.cc ../ffdhe.cc ../certificate_algorithms.cc ../credential_requests.cc ../integrity_only.cc ../native_provider.cc ../transparency.cc ../dnssec.cc ../its_credentials.cc)
target_compile_definitions(httpps_native_test PRIVATE PHROMIUM_HTTPPS_STANDALONE PHROMIUM_HTTPPS_TEST_DIAGNOSTICS)
target_include_directories(httpps_native_test PRIVATE "${CHROMIUM_ROOT}")
target_link_libraries(httpps_native_test PRIVATE crypto zlibstatic httpps_brotli libzstd_static ${CMAKE_DL_LIBS})
add_executable(httpps_pki_test pki_test.cc)
target_include_directories(httpps_pki_test PRIVATE "${CHROMIUM_ROOT}")
target_link_libraries(httpps_pki_test PRIVATE pki)
add_executable(httpps_pki_chain_test pki_chain_test.cc)
target_include_directories(httpps_pki_chain_test PRIVATE "${CHROMIUM_ROOT}")
target_link_libraries(httpps_pki_chain_test PRIVATE pki)
add_executable(httpps_transparency_test transparency_test.cc ../transparency.cc)
target_compile_definitions(httpps_transparency_test PRIVATE PHROMIUM_HTTPPS_STANDALONE)
target_include_directories(httpps_transparency_test PRIVATE "${CHROMIUM_ROOT}")
target_link_libraries(httpps_transparency_test PRIVATE crypto)
if(WIN32)
  target_link_libraries(httpps_native_test PRIVATE ws2_32)
endif()
enable_testing()
add_test(NAME tls20_primitives COMMAND httpps_native_test)
add_test(NAME httpps_spki_and_https_policy COMMAND httpps_pki_test)
add_test(NAME httpps_pki_path_builder COMMAND httpps_pki_chain_test)
add_test(NAME httpps_ct_v2_provider COMMAND httpps_transparency_test)
add_executable(httpps_dnssec_test dnssec_test.cc ../dnssec.cc)
target_compile_definitions(httpps_dnssec_test PRIVATE PHROMIUM_HTTPPS_STANDALONE)
target_include_directories(httpps_dnssec_test PRIVATE "${CHROMIUM_ROOT}")
target_link_libraries(httpps_dnssec_test PRIVATE crypto ${CMAKE_DL_LIBS})
add_test(NAME httpps_dnssec_policy COMMAND httpps_dnssec_test)
add_executable(httpps_its_test its_credentials_test.cc ../its_credentials.cc)
target_compile_definitions(httpps_its_test PRIVATE PHROMIUM_HTTPPS_STANDALONE)
target_include_directories(httpps_its_test PRIVATE "${CHROMIUM_ROOT}")
target_link_libraries(httpps_its_test PRIVATE crypto ${CMAKE_DL_LIBS})
add_test(NAME httpps_its_policy COMMAND httpps_its_test)
